```python
def get_effective_key(cs_config: dict) -> tuple:
    """CS配置から実効パラメータキーを生成"""
    key_list = []
    for csid, ports, cap in zip(cs_config['csids'], cs_config['ports'], cs_config['cap_kw']):
        if ports > 0:
            key_list.append((csid, ports, cap))
        else:
            key_list.append((csid, 0, -1))
    return tuple(key_list)
```
